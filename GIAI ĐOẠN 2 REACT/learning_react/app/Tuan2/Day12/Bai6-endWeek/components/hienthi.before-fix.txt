import  styles  from "./hienthi.module.css";
export default function HienThiNasaItem(props: {
  description?: string;
  title: string;
  date_created: string;
  mediaImg?: string;
  nasa_id: string;
  like:boolean;
  save:boolean;
  onLike:() => void;
  onSave:() => void
}) {
  return (
    <div className="Item-Nasa">
      <div className="box-img"><img src={props.mediaImg} alt="Ảnh" width="300px" /></div>
      <div className="box-description">
          <h1>Tiêu đề :{props.title}</h1>
          <p>
            ID của Nasa:{props.nasa_id}
            Mô tả:{props.description}
            Ngày tạo:{new Date(props.date_created).toLocaleDateString("vi-VN")};
          </p>
          
      </div>
      <div className="box-footer">
        <button onClick={props.onLike} className="btn btn-like">{props.like? "Đã tim" :"Tim"}</button>
        <button onClick={props.onSave} className="btn btn-save">{props.save? "Đã lưu" :"Lưu" }</button>
      </div>
    </div>
  );
}
